  <meta>
    <profile
             value="http://fhir.hcen.gub.uy/StructureDefinition/hcen-practitioner"/>
  </meta>
  <language value="es"/>
  <text>
    <status value="generated"/>
    <div xml:lang="es" xmlns="http://www.w3.org/1999/xhtml" lang="es"><p>Lucía Silva. Cédula de ejemplo 12345672.</p></div>
  </text>
  <identifier>
    <type>
      <coding>
        <system value="http://terminology.hl7.org/CodeSystem/v2-0203"/>
        <code value="NI"/>
        <display value="National unique individual identifier"/>
      </coding>
    </type>
    <system value="urn:oid:2.16.858.2.10000675.68909"/>
    <value value="12345672"/>
  </identifier>
  <name>
    <family value="Silva"/>
    <given value="Lucía"/>
  </name>
</Practitioner>
